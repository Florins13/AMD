-- complete the following view in order to build
-- the output data presented below at [OUT]
-- CREATE VIEW v1( VC1, VC2, VC3 ) AS
-- ;

CREATE VIEW v1( VC1, VC2, VC3 ) AS
select R2.C1 AS VC1, R1.C1 AS VC2, R3.C1 AS VC3
from R2
inner join R1 ON R2.C1_R1 = R1.C1
inner join R3 ON R3.C1_R1 = R1.C1
order by R3.C1
 ;



-- after completing the previous code uncomment the following 2 lines
SELECT VC1, VC2, VC3
FROM v1
;



--==========================

--| vc1 | vc2 |    vc3
--|-----+-----+------------
--| 111 |   1 | 2012-01-31
--| 111 |   1 | 2012-02-28
--| 222 |   2 | 2012-03-31
--|(3 rows)

-- view v1_domain (nomes das colunas)
CREATE VIEW v1_domain( vc1, vc2, vc3 ) AS
SELECT
    'vc1'::text AS vc1,
    'vc2'::text AS vc2,
    'vc3'::text AS vc3
;

-- view v1_attrType (tipos de atributos)
CREATE VIEW v1_attrType( vc1, vc2, vc3 ) AS
SELECT
    'discrete'::text AS vc1,
    'discrete'::text AS vc2,
    'discrete'::text AS vc3
;

-- view v1_class (marca a coluna de classe)
CREATE VIEW v1_class( vc1, vc2, vc3 ) AS
SELECT
    NULL::text    AS vc1,
    NULL::text    AS vc2,
    'class'::text AS vc3
;



-- view v1_dataset (3-Row-Header format)
DROP VIEW IF EXISTS v1_dataset;

CREATE VIEW v1_dataset( my_order, vc1, vc2, vc3 ) AS
SELECT *
FROM (
    -- linha 1: nomes das colunas
    SELECT 0 AS my_order, vc1, vc2, vc3
    FROM v1_domain

    UNION ALL

    -- linha 2: tipos dos atributos
    SELECT 1 AS my_order, vc1, vc2, vc3
    FROM v1_attrType

    UNION ALL

    -- linha 3: coluna de classe
    SELECT 2 AS my_order, vc1, vc2, vc3
    FROM v1_class

    UNION ALL

    -- linhas seguintes: dados reais
    SELECT 3 AS my_order, VC1::text, VC2::text, VC3::text
    FROM v1
) AS combined
ORDER BY my_order
;

-- Test the dataset view
SELECT VC1, VC2, VC3 FROM v1_dataset ORDER BY my_order;

-- Generate text file 
\COPY (SELECT vc1, vc2, vc3 FROM v1_dataset ORDER BY my_order) TO 'dataset_3RowHeader.txt' WITH (FORMAT csv, HEADER false, DELIMITER ',');

-- Follow the "practiceGuide.pdf" until arriving at this goal:
-- create the "dataset view" compatible with the Orange "Header-with-Attribute-Type" format

-- view v2_attrType (nomes com prefixo de tipo/role)
CREATE VIEW v2_attrType( vc1, vc2, vc3 ) AS
SELECT
    'D#vc1'::text  AS vc1,   -- Discrete
    'D#vc2'::text  AS vc2,   -- Discrete
    'Dc#vc3'::text AS vc3    -- Discrete + class
;

-- view v2_dataset (Header-with-Attribute-Type format)
DROP VIEW IF EXISTS v2_dataset;

CREATE VIEW v2_dataset( vc1, vc2, vc3 ) AS
SELECT vc1, vc2, vc3
FROM v2_attrType

UNION ALL

SELECT VC1::text, VC2::text, VC3::text
FROM v1
;

-- Test the v2_dataset view
SELECT vc1, vc2, vc3 FROM v2_dataset;

-- Generate text file for representation of v2_dataset
\COPY (SELECT vc1, vc2, vc3 FROM v2_dataset) TO 'dataset_HeaderWithType.txt' WITH (FORMAT csv, HEADER false, DELIMITER ',');
